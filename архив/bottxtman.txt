func loadConfigFromViper() {
	err := viper.Unmarshal(&config)
	if err != nil {
		log.Fatal(err)
	}
	logsVerbose = viper.GetString("logs_verbose")
	tgToken = viper.GetString("tg_token")
	tgChannelID = viper.GetInt64("tg_channel")
	preAcl := viper.GetStringSlice("tg_acl")
	tgAcl = make([]int64, 0)
	for _, v := range preAcl {
		id, err := strconv.ParseInt(v, 10, 64)
		if err != nil {
			log.Fatal(err)
		}
		tgAcl = append(tgAcl, id)
	}
	proxyRequestTimeout = viper.GetDuration("proxyRequestTimeout")
	configReloadEnabled = viper.GetBool("config_reload_enabled")
	autoChecksEnable = viper.GetBool("auto_checks_enable")
	onAddChecksEnable = viper.GetBool("on_add_checks_enable")
	logsDir = viper.GetString("logs_dir")
	proxyChecksDefaultUnmarshal := viper.Get("proxy_checks.default").([]interface{})
	for _, value := range proxyChecksDefaultUnmarshal {
		var proxyCheck ConfigProxyCheck
		data, _ := json.Marshal(value)
		json.Unmarshal(data, &proxyCheck)
		proxyChecksDefault = append(proxyChecksDefault, proxyCheck)
	}
	//	reloadPeriod = viper.Get("reloadEnabled").(time.Time)
	rawProxyFiles := viper.Get("proxy_files")

	proxyFiles = make([]ConfigProxyFiles, 0)
	if rawProxyFiles != nil {
		if proxyFileSlice, ok := rawProxyFiles.([]interface{}); ok {
			for _, proxyFileItem := range proxyFileSlice {
				if proxyFileItemMap, ok := proxyFileItem.(map[string]interface{}); ok {
					var proxyFile ConfigProxyFiles
					proxyFile.File = proxyFileItemMap["file"].(string)
					proxyFile.Country = proxyFileItemMap["country"].(string)
					proxyFiles = append(proxyFiles, proxyFile)

				}
			}
		}
	}
	// fmt.Print(proxyFiles)
}

										msg := tgbotapi.NewMessage(update.Message.Chat.ID, fmt.Sprintf("Country: %s\nProxy: %s\nState: [FAIL] Not Added\nReason: Incorrect Proxy format", country, line))
										_, err = bot.Send(msg)
										if err != nil {
											log.Println(err)
										}
										continue
									}
									proxyAddr := proxyPart[0]
									proxyPort := proxyPart[1]
									proxyLogin := proxyPart[2]
									proxyPass := proxyPart[3]
									line = fmt.Sprintf("%s://%s:%s@%s:%s", sepProxy[0], proxyLogin, proxyPass, proxyAddr, proxyPort)
								} else if format == "3" {
									line = fmt.Sprintf("socks5h://%s", line)
								}

								if onAddChecksEnable {
									countryChecked, err := getCountry(line)
									if err != nil {
										msg := tgbotapi.NewMessage(update.Message.Chat.ID, fmt.Sprintf("Country: %s\nProxy: %s\nState: [FAIL] Not Added\nReason: Proxy check failed: No answer from site via proxy", countryChecked, line))
										_, err = bot.Send(msg)
										if err != nil {
											log.Println(err)
										}
										continue
									}
									if country != countryChecked {
										msg := tgbotapi.NewMessage(update.Message.Chat.ID, fmt.Sprintf("Country: %s\nProxy: %s\nState: [FAIL] Not Added\nReason: Proxy check failed: Different Country: %s (Must be %s)", country, line, countryChecked, country))
										_, err = bot.Send(msg)
										if err != nil {
											log.Println(err)
										}
										continue
									}
								}

								// вносим строку в файл
								err = addProxyToFile(proxyFile, line)
								if err != nil {
									log.Println(err)
								}

								msg := tgbotapi.NewMessage(update.Message.Chat.ID, fmt.Sprintf("Country: %s\nProxy: %s\nState: Added", country, line))
								_, err = bot.Send(msg)
								if err != nil {
									log.Println(err)
								}
							}
						}
					}
					if action == "set" {
						if checkListContainsString(countriesList, country) == true {
							var newLines []string
							for _, line := range strings.Split(strings.TrimSuffix(update.Message.Text, "\n"), "\n") {
								newLines = append(newLines, line)
								_, err := url.Parse(line)
								if err != nil {
									log.Println(err)
								}
							}

							err, oldLines := setProxyInFile(proxyFile, newLines, format, country)
							if err != nil {
								log.Println(err)
							}
							for _, line := range oldLines {
								msg := tgbotapi.NewMessage(update.Message.Chat.ID, fmt.Sprintf("Country: %s\nProxy: %s\nState: Deleted", country, line))
								_, err = bot.Send(msg)
								if err != nil {
									log.Println(err)
								}

							}
							for _, line := range newLines {
								msg := tgbotapi.NewMessage(update.Message.Chat.ID, fmt.Sprintf("Country: %s\nProxy: %s\nState: Added", country, line))
								_, err = bot.Send(msg)
								if err != nil {
									log.Println(err)
								}

							}
						}
					}
					if action == "check" {
						if checkListContainsString(countriesList, country) == true {
							var newLines []string
							for _, line := range strings.Split(strings.TrimSuffix(update.Message.Text, "\n"), "\n") {
								newLines = append(newLines, line)
								_, err := url.Parse(line)
								if err != nil {
									log.Println(err)
								}
							}

							err, oldLines := setProxyInFile(proxyFile, newLines, format, country)
							if err != nil {
								log.Println(err)
							}
							for _, line := range oldLines {
								msg := tgbotapi.NewMessage(update.Message.Chat.ID, fmt.Sprintf("Country: %s\nProxy: %s\nState: Deleted", country, line))
								_, err = bot.Send(msg)
								if err != nil {
									log.Println(err)
								}

							}
							for _, line := range newLines {
								msg := tgbotapi.NewMessage(update.Message.Chat.ID, fmt.Sprintf("Country: %s\nProxy: %s\nState: Added", country, line))
								_, err = bot.Send(msg)
								if err != nil {
									log.Println(err)
								}

							}
						}
					}
				}
			}
		} else if update.CallbackQuery != nil {
			callback := tgbotapi.NewCallback(update.CallbackQuery.ID, update.CallbackQuery.Data)
			if _, err := bot.Request(callback); err != nil {
				panic(err)
			}

			parts := strings.Split(update.CallbackQuery.Message.Text, "\n")

			partsProxy := strings.Split(parts[1], "Proxy: ")
			partsCounty := strings.Split(parts[2], "Country: ")
			fmt.Printf("proxy - %s\n", partsProxy[1])
			fmt.Printf("country - %s\n", partsCounty[1])

			if checkListContainsString(countriesList, partsCounty[1]) == true {
				for _, file := range proxyFiles {
					if file.Country == partsCounty[1] {
						proxies, err := getProxiesFromFile(file.File)
						if err != nil {
							log.Println(err)
						}
						if checkListContainsString(proxies, partsProxy[1]) == true {
							deleteProxyFromFile(file.File, partsProxy[1])
						}
						proxies, err = getProxiesFromFile(file.File)
						msg := tgbotapi.NewMessage(update.CallbackQuery.Message.Chat.ID, fmt.Sprintf("Country: %s\nProxy: %s\nState: Deleted\nMade by User: %s", file.Country, partsProxy[1], update.CallbackQuery.From.UserName))

						_, err = bot.Send(msg)
						if err != nil {
							log.Println(err)
						}
					}
				}
			}
		}
	}
}

func manipulatorGoProxiesChecks() {
	log.Debug("manipulatorGoProxiesChecks started")
	log.Tracef("ProxyFiles: %s", proxyFiles)
	for {
		log.Tracef("manipulatorGoProxiesChecks new cycle")
		proxiesChecksGoToDelete := make([]ProxyGoroutine, len(proxyGoroutineList))
		copy(proxiesChecksGoToDelete, proxyGoroutineList)
		log.Tracef("proxiesChecksGoToDelete before: %v", proxiesChecksGoToDelete)
		for _, singleProxyFile := range proxyFiles {

			proxies, err := readProxies(singleProxyFile.File)
			if err != nil {
				log.Error("Failed to read proxies", err)
				continue
			}
			log.Tracef("Proxy File for country %s readed succesfully (%s)", singleProxyFile.Country, singleProxyFile.File)
			for _, proxy := range proxies {
				exists := searchProxyInProxyGoroutinesList(singleProxyFile.Country, proxy)
				if !exists {
					log.Infof("Added to proxyGoroutineList Country: %s Url: %s", singleProxyFile.Country, proxy)
					goRoutine := ProxyGoroutine{
						Country:  singleProxyFile.Country,
						Url:      proxy,
						StopChan: make(chan struct{}),
					}
					proxyGoroutineList = append(proxyGoroutineList, goRoutine)
					log.Debugf("List of proxies right now: %v", proxyGoroutineList)

					go func(goRoutine ProxyGoroutine) {
						var unsuccesfulChecksCount int64
						var differentCountryCount int64
						unsuccesfulChecksCount = 0
						differentCountryCount = 0
						for {
							select {
							case <-goRoutine.StopChan:
								log.Infof("Closed Checking Goroutine for proxy: %s Url: %s", goRoutine.Country, goRoutine.Url)
								return
							default:
								// вычисляем действия, описанные вами
								log.Tracef("Cycle check started Country: %s Proxy: %s", goRoutine.Country, goRoutine.Url)
								checkSuccessful := false
								countrySame := true
								countryResponse := ""
								for i, c := range proxyChecksDefault {
									check := proxyCheck(goRoutine.Url, c.Url, c.GetCountry)
									if check.Error != nil {
										log.Debugf("[%d] check failed (%s). Err: %s Country: %s Proxy %s", i, c.Url, check.Error, goRoutine.Country, goRoutine.Url)
										logFileUnsuccess.Infof("[%d] check failed! (%s) (Checked via %s). Country: %s Proxy %s",
											i, check.Country,
											c.Url,
											goRoutine.Country,
											goRoutine.Url)
									}
									if check.Country == "" && check.Error != nil {
										log.Tracef("[%d] Country not checked (%s). Country: %s Proxy %s", i, c.Url, goRoutine.Country, goRoutine.Url)
									} else if check.Country == "!" && check.Error != nil {
										log.Tracef("[%d] Country check error (!) (%s). Country: %s Proxy %s", i, c.Url, goRoutine.Country, goRoutine.Url)
									} else if check.Country != goRoutine.Country && check.Error != nil {
										log.Tracef("[%d] Country check is different! (%s) (Checked via %s). Country: %s Proxy %s", i, check.Country, c.Url, goRoutine.Country, goRoutine.Url)
										countrySame = false
										countryResponse = check.Country
										logFileUnsuccess.Infof("[%d] Country check is different! (%s) (Checked via %s). Country: %s Proxy %s",
											i, check.Country,
											c.Url,
											goRoutine.Country,
											goRoutine.Url)
									}
									if check.Available == true {
										checkSuccessful = true
										log.Tracef("[%d] check succesful (%s). No more checks needed. Country: %s Proxy %s", i, c.Url, goRoutine.Country, goRoutine.Url)
										triggerToAdd := true
										for _, OKGoRoutine := range currentOK {
											if goRoutine.Url == OKGoRoutine.Url && goRoutine.Country == OKGoRoutine.Country {
												triggerToAdd = false
											}
										}
										if triggerToAdd {
											currentOK = append(currentOK, goRoutine)
										}
										break
									}
								}

								if !countrySame {
									differentCountryCount++
								} else {
									if differentCountryCount >= 3 {
										msg := tgbotapi.NewMessage(
											tgChannelID,
											fmt.Sprintf("[COUNTRY FIXED] Proxy shows again correct country. Number of tries with incorrect country: %d.\nProxy: %s\nCountry: %s",
												differentCountryCount, goRoutine.Url, goRoutine.Country))
										if _, err := bot.Send(msg); err != nil {
											log.Error("Failed to send message", err)
										}
										for i, wrongGoRoutine := range currentDifferentCountry {
											if goRoutine.Url == wrongGoRoutine.Url && goRoutine.Country == wrongGoRoutine.Country {
												currentDifferentCountry = append(currentDifferentCountry[:i], currentDifferentCountry[i+1:]...)
											}
										}
									}
									differentCountryCount = 0
								}

								if !checkSuccessful {
									unsuccesfulChecksCount++
								} else {
									if unsuccesfulChecksCount >= 3 {
										filePath, found := findFileByCountry(goRoutine.Country)
										if !found {
											log.Warnf("Country %s not found in proxyFiles", goRoutine.Country)
											continue
										}

										err := outautoquarantineProxyFromFile(filePath, goRoutine.Url)
										if err != nil {
											log.Error("Failed to move proxy from quarantine", err)
										}

										msg := tgbotapi.NewMessage(
											tgChannelID,
											fmt.Sprintf("[ALIVE] Proxy became alive. Number of unsuccessful tries: %d.\nProxy: %s\nCountry: %s",
												unsuccesfulChecksCount, goRoutine.Url, goRoutine.Country))
										if _, err := bot.Send(msg); err != nil {
											log.Error("Failed to send message", err)
										}
										for i, unavailableGoRoutine := range currentUnavailable {
											if goRoutine.Url == unavailableGoRoutine.Url && goRoutine.Country == unavailableGoRoutine.Country {
												currentUnavailable = append(currentUnavailable[:i], currentUnavailable[i+1:]...)
											}
										}
									}

									unsuccesfulChecksCount = 0
								}

								if unsuccesfulChecksCount == 1 || differentCountryCount == 1 {
									time.Sleep(1 * time.Minute) // подождать 1 минуту после первой попытки
								} else if unsuccesfulChecksCount == 2 || differentCountryCount == 1 {
									time.Sleep(2 * time.Minute) // подождать 2 минуты после второй попытки
								} else if unsuccesfulChecksCount == 3 {

									filePath, found := findFileByCountry(goRoutine.Country)
									if !found {
										log.Warnf("Country %s not found in proxyFiles", goRoutine.Country)
										continue
									}

									err := setautoquarantineProxyFromFile(filePath, goRoutine.Url)
									if err != nil {
										log.Error("Failed to move proxy to quarantine", err)
									}

									currentUnavailable = append(currentUnavailable, goRoutine)

									for i, okRoutine := range currentOK {
										if goRoutine.Url == okRoutine.Url && goRoutine.Country == okRoutine.Country {
											currentOK = append(currentOK[:i], currentOK[i+1:]...)
										}
									}

									inlineKeyboard := tgbotapi.NewInlineKeyboardMarkup(
										tgbotapi.NewInlineKeyboardRow(
											tgbotapi.NewInlineKeyboardButtonData("remove", "remove"),
										),
									)
									msg := tgbotapi.NewMessage(
										tgChannelID,
										fmt.Sprintf("[DEAD] Check failed.\nProxy: %s\nCountry: %s", goRoutine.Url, goRoutine.Country))
									msg.ReplyMarkup = inlineKeyboard
									if _, err := bot.Send(msg); err != nil {
										log.Error("Failed to send message", err)
									}
								} else if differentCountryCount == 3 {
									currentDifferentCountry = append(currentDifferentCountry, goRoutine)

									for i, okRoutine := range currentOK {
										if goRoutine.Url == okRoutine.Url && goRoutine.Country == okRoutine.Country {
											currentOK = append(currentOK[:i], currentOK[i+1:]...)
										}
									}

									inlineKeyboard := tgbotapi.NewInlineKeyboardMarkup(
										tgbotapi.NewInlineKeyboardRow(
											tgbotapi.NewInlineKeyboardButtonData("remove", "remove"),
										),
									)
									msg := tgbotapi.NewMessage(
										tgChannelID,
										fmt.Sprintf("[COUNTRY DIFFERENT] Different country! (%s).\nProxy: %s\nCountry: %s", countryResponse, goRoutine.Url, goRoutine.Country))
									msg.ReplyMarkup = inlineKeyboard
									if _, err := bot.Send(msg); err != nil {
										log.Error("Failed to send message", err)
									}

								} else {
									time.Sleep(10 * time.Minute)
								}
							}
						}
					}(goRoutine)
				} else {
					for i, g := range proxiesChecksGoToDelete {
						if g.Country == singleProxyFile.Country && g.Url == proxy {
							proxiesChecksGoToDelete = append(proxiesChecksGoToDelete[:i], proxiesChecksGoToDelete[i+1:]...)
						}
					}
				}

			}
		}
		log.Tracef("proxiesChecksGoToDelete after: %v", proxiesChecksGoToDelete)
		if len(proxiesChecksGoToDelete) > 0 {
			for _, pcd := range proxiesChecksGoToDelete {
				for i, pr := range currentOK {
					if pr.Country == pcd.Country && pr.Url == pcd.Url {
						log.Tracef("Removing from okList Country: %s Url: %s", pr.Country, pr.Url)
						currentOK = append(currentOK[:i], currentOK[i+1:]...)
					}
				}
				for i, pr := range currentUnavailable {
					if pr.Country == pcd.Country && pr.Url == pcd.Url {
						log.Tracef("Removing from okList Country: %s Url: %s", pr.Country, pr.Url)
						currentOK = append(currentUnavailable[:i], currentUnavailable[i+1:]...)
					}
				}
				for i, pr := range currentDifferentCountry {
					if pr.Country == pcd.Country && pr.Url == pcd.Url {
						log.Tracef("Removing from okList Country: %s Url: %s", pr.Country, pr.Url)
						currentOK = append(currentDifferentCountry[:i], currentDifferentCountry[i+1:]...)
					}
				}
				for i, pg := range proxyGoroutineList {
					if pg.Country == pcd.Country && pg.Url == pcd.Url {
						log.Tracef("Removing from proxyGoroutineList Country: %s Url: %s", pg.Country, pg.Url)
						proxyGoroutineList = append(proxyGoroutineList[:i], proxyGoroutineList[i+1:]...)
						close(pg.StopChan)
					}
				}

			}
		}
		time.Sleep(10 * time.Second)
	}
}

func setProxyInFile(f string, l []string, format string, c string) (error, []string) {
	data, err := os.ReadFile(f)
	if err != nil {
		return err, nil
	}

	// разбиваем файл на строки
	lines := strings.Split(string(data), "\n")
	var clearedLines []string
	var peerLines []string
	for _, line := range lines {
		if !strings.HasPrefix(line, "peer ") {
			clearedLines = append(clearedLines, line)
		} else {
			peerLines = append(peerLines, line)
		}
	}

	// удаляем строку, которую хотим удалить
	for _, proxy := range l {

		if format == "1" {
			proxyPart := strings.Split(proxy, ":") //addr:port:login:pass
			if len(proxyPart) < 4 {
				return errors.New("неверный формат прокси: ожидалось 4 части"), nil
			}

			proxyAddr := proxyPart[0]
			proxyPort := proxyPart[1]
			proxyLogin := proxyPart[2]
			proxyPass := proxyPart[3]

			proxy = fmt.Sprintf("socks5h://%s:%s@%s:%s", proxyLogin, proxyPass, proxyAddr, proxyPort)
		} else if format == "2" {
			sepProxy := strings.SplitN(proxy, "://", 2)
			proxyPart := strings.Split(sepProxy[1], ":") //addr:port:login:pass
			if len(proxyPart) < 4 {
				return errors.New("неверный формат прокси: ожидалось 4 части"), nil
			}
			proxyAddr := proxyPart[0]
			proxyPort := proxyPart[1]
			proxyLogin := proxyPart[2]
			proxyPass := proxyPart[3]
			proxy = fmt.Sprintf("%s://%s:%s@%s:%s", sepProxy[0], proxyLogin, proxyPass, proxyAddr, proxyPort)
		} else if format == "3" {
			proxy = fmt.Sprintf("socks5h://%s", proxy)
		}

		country, err := getCountry(proxy)
		if err != nil {
			return errors.New("Proxy check failed: No answer from site via proxy"), nil
		}
		if country != c {
			return errors.New(fmt.Sprintf("Proxy check failed: Different country: %s (Must be %s)", country, c)), nil
		}

		clearedLines = append(clearedLines, fmt.Sprintf("peer %s", proxy))
	}
	// записываем новые строки обратно в файл
	err = os.WriteFile(f, []byte(strings.Join(clearedLines, "\n")), 0o644)
	if err != nil {
		return err, nil
	}

	return nil, peerLines
}

func addProxyToFile(f string, p string) error {
	file, err := os.OpenFile(f, os.O_APPEND|os.O_WRONLY|os.O_CREATE, 0o600)
	if err != nil {
		return err
	}

	defer file.Close()

	if _, err = file.WriteString(fmt.Sprintf("peer %s\n", p)); err != nil {
		return err
	}

	return nil
}

func deleteProxyFromFile(f string, p string) error {
	// читаем файл
	data, err := os.ReadFile(f)
	if err != nil {
		return err
	}

	// разбиваем файл на строки
	lines := strings.Split(string(data), "\n")

	// удаляем строку, которую хотим удалить
	var newLines []string
	var peerDeleted bool
	for _, line := range lines {
		if line != fmt.Sprintf("peer %s", p) || peerDeleted == true {
			newLines = append(newLines, line)
		} else {
			peerDeleted = true
		}
	}

	// записываем новые строки обратно в файл
	err = os.WriteFile(f, []byte(strings.Join(newLines, "\n")), 0o644)
	if err != nil {
		return err
	}

	return nil
}

func deleteQuarantineProxyFromFile(f string, p string) error {
	// Читаем файл
	data, err := os.ReadFile(f)
	if err != nil {
		return err
	}

	// Разбиваем на строки
	lines := strings.Split(string(data), "\n")

	var newLines []string
	var deleted bool // Флаг удаления — заменяет peerDeleted для ясности

	for _, line := range lines {
		line = strings.TrimSpace(line) // Убираем лишние пробелы и переносы

		// Проверяем, начинается ли строка с quarantine + proxy
		if !deleted && strings.HasPrefix(line, "quarantine "+p) {
			deleted = true // Пропускаем эту строку
			continue
		}

		// Добавляем все остальные строки
		newLines = append(newLines, line)
	}

	// Перезаписываем файл
	err = os.WriteFile(f, []byte(strings.Join(newLines, "\n")), 0o644)
	if err != nil {
		return err
	}

	return nil
}

func getProxiesFromFile(filePath string) ([]string, error) {
	data, err := os.ReadFile(filePath)
	if err != nil {
		return nil, err
	}

	lines := strings.Split(string(data), "\n")
	var result []string

	for _, line := range lines {
		line = strings.TrimSpace(line) // Убираем лишние пробелы по краям
		if line == "" {
			continue // Пропускаем пустые строки
		}

		if strings.HasPrefix(line, "peer ") {
			// Убираем префикс "peer "
			cleanLine := strings.TrimSpace(strings.TrimPrefix(line, "peer "))
			result = append(result, cleanLine)
		}
		if strings.HasPrefix(line, "quarantine ") {
			cleanProxy := strings.TrimSpace(strings.TrimPrefix(line, "quarantine "))
			result = append(result, fmt.Sprintf("QUARANTINE %s", cleanProxy))
		}
		if strings.HasPrefix(line, "autoquarantine ") {
			cleanProxy := strings.TrimSpace(strings.TrimPrefix(line, "autoquarantine "))
			result = append(result, fmt.Sprintf("AUTOQUARANTINE %s", cleanProxy))
		}

	}

	return result, nil
}

func checkProxies(bot *tgbotapi.BotAPI, channelID int64, configs []ConfigProxyFiles) {
	ticker := time.NewTicker(10 * time.Minute)
	defer ticker.Stop()

	for range ticker.C {
		for _, config := range configs {
			proxies, err := readProxies(config.File)
			if err != nil {
				log.Error("Failed to read proxies in checker", err)
				continue
			}
			for _, proxy := range proxies {
				// Check country change
				country, err := getCountry(proxy)
				if err != nil {
					log.Error("[DEAD] Check failed to get country", err)

					inlineKeyboard := tgbotapi.NewInlineKeyboardMarkup(
						tgbotapi.NewInlineKeyboardRow(
							tgbotapi.NewInlineKeyboardButtonData("remove", "remove"),
						),
					)
					msg := tgbotapi.NewMessage(channelID, fmt.Sprintf("[DEAD] Check failed.\nProxy: %s\nCountry: %s", proxy, config.Country))

					msg.ReplyMarkup = inlineKeyboard
					if _, err := bot.Send(msg); err != nil {
						log.Error("Failed to send message", err)
					}
					continue
				} else {
					log.Debug(fmt.Sprintf("[Alive] Check succesfull. Proxy: %s , Country %s ", proxy, config.Country))
				}

				if country != config.Country {
					log.Error(fmt.Sprintf("[Country] Check failed. Country is different. Proxy: %s , Country: %s , Proxy Country: %s", proxy, config.Country, country))
					// Notify users

					inlineKeyboard := tgbotapi.NewInlineKeyboardMarkup(
						tgbotapi.NewInlineKeyboardRow(
							tgbotapi.NewInlineKeyboardButtonData("remove", "remove"),
						),
					)

					msg := tgbotapi.NewMessage(channelID, fmt.Sprintf("Proxy %s changed country from %s to %s!", proxy, config.Country, country))
					msg.ReplyMarkup = &inlineKeyboard

					if _, err := bot.Send(msg); err != nil {
						log.Error("Failed to send message", err)
					}

				} else {
					log.Debug(fmt.Sprintf("[Alive] Check succesfull. Proxy: %s , Country %s , Proxy Country %s", proxy, config.Country, country))
				}
			}
		}
	}
}

func readProxies(filename string) ([]string, error) {
	file, err := os.Open(filename)
	if err != nil {
		return nil, err
	}
	defer file.Close()

	var proxies []string
	scanner := bufio.NewScanner(file)
	for scanner.Scan() {
		line := scanner.Text()
		if strings.HasPrefix(line, "peer") {
			proxies = append(proxies, strings.TrimSpace(line[5:]))
		}
		if strings.HasPrefix(line, "autoquarantine") {
			proxies = append(proxies, strings.TrimSpace(line[15:]))
		}
	}

	return proxies, scanner.Err()
}

func proxyCheck(proxy string, checkUrl string, getCountry bool) ProxyCheckResponse {
	a := ProxyCheckResponse{Available: false, Country: "", Error: nil, Body: ""}

	proxyUrl, err := url.Parse(proxy)
	if err != nil {
		a.Error = err
		return a
	}
	transport := &http.Transport{Proxy: http.ProxyURL(proxyUrl)}
	client := &http.Client{Transport: transport}

	resp, err := client.Get(checkUrl)
	if err != nil {
		a.Error = err
		return a
	}
	defer resp.Body.Close()

	if resp.StatusCode == http.StatusOK || resp.StatusCode == http.StatusPermanentRedirect || resp.StatusCode == http.StatusTemporaryRedirect {
		a.Available = true
	} else {
		a.Error = err
		return a
	}
	body, err := io.ReadAll(resp.Body)
	if err != nil {
		a.Error = err
		return a
	}
	a.Body = string(body)

	if getCountry {
		var data map[string]interface{}
		err = json.NewDecoder(resp.Body).Decode(&data)
		if err != nil {
			a.Error = err
			return a
		}

		country, ok := data["country"].(string)
		if !ok {
			a.Error = fmt.Errorf("country field not found in JSON response")
			return a
		} else {
			a.Country = country
		}
	}

	return a
}

func getCountry(proxy string) (string, error) {
	time.Sleep(2 * time.Second)
	// Get country from ipinfo.io
	proxyUrl, err := url.Parse(proxy)
	if err != nil {
		return "[func getCountry] Can't parse proxy url", err
	}

	transport := &http.Transport{Proxy: http.ProxyURL(proxyUrl)}
	client := &http.Client{Transport: transport}

	resp, err := client.Get("https://ipinfo.io/")
	if err != nil {
		return "", err
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return "", err
	}

	var data map[string]interface{}
	err = json.NewDecoder(resp.Body).Decode(&data)
	if err != nil {
		return "", err
	}

	country, ok := data["country"].(string)
	if !ok {
		return "", fmt.Errorf("country field not found in JSON response")
	}

	return country, nil
}

func removeProxyFromFile(filename string, proxy string) error {

	file, err := os.Open(filename)
	if err != nil {
		return err
	}
	defer file.Close()

	var lines []string
	scanner := bufio.NewScanner(file)
	for scanner.Scan() {
		line := scanner.Text()
		if !strings.HasPrefix(line, "peer "+proxy) {
			lines = append(lines, line)
		}
	}

	err = os.Truncate(filename, 0)
	if err != nil {
		return err
	}

	file, err = os.OpenFile(filename, os.O_WRONLY, 0o644)
	if err != nil {
		return err
	}
	defer file.Close()

	for _, line := range lines {
		_, err := file.WriteString(line + "\n")
		if err != nil {
			return err
		}
	}

	return nil
}

func setautoquarantineProxyFromFile(filename string, proxy string) error {
	// Открываем файл для чтения
	file, err := os.Open(filename)
	if err != nil {
		return err
	}
	defer file.Close()

	var lines []string
	scanner := bufio.NewScanner(file)

	for scanner.Scan() {
		line := scanner.Text()

		// Проверяем, начинается ли строка с "peer " и содержит ли она нужный прокси
		if strings.HasPrefix(line, "peer ") {
			existingProxy := strings.TrimSpace(strings.TrimPrefix(line, "peer "))
			if existingProxy == proxy {
				// Заменяем "peer" на "autoquarantine"
				line = strings.Replace(line, "peer ", "autoquarantine ", 1)
			}
		}

		lines = append(lines, line)
	}

	// Перезаписываем файл
	err = os.WriteFile(filename, []byte(strings.Join(lines, "\n")+"\n"), 0644)
	if err != nil {
		return err
	}

	return nil
}

func splitMessageByNewline(message string) []string {
	var messages []string
	lines := strings.Split(message, "\n") // Разбиваем текст на строки
	currentMessage := ""
	currentLength := 0

	for _, line := range lines {
		lineLength := len([]rune(line)) + 1 // +1 для учета \n

		// Если добавление строки превысит лимит
		if currentLength+lineLength > 4000 {
			if currentMessage != "" {
				messages = append(messages, strings.TrimSpace(currentMessage))
			}
			currentMessage = line // Начинаем новое сообщение с текущей строки
			currentLength = lineLength
		} else {
			// Добавляем строку к текущему сообщению
			if currentMessage != "" {
				currentMessage += "\n"
			}
			currentMessage += line
			currentLength += lineLength
		}
	}

	// Добавляем последнее сообщение, если оно не пустое
	if currentMessage != "" {
		messages = append(messages, strings.TrimSpace(currentMessage))
	}

	return messages
}
